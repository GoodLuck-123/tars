cmake_minimum_required(VERSION 3.10)
project(livo_flir_camera)
set(CMAKE_CXX_STANDARD 14)
find_package(catkin REQUIRED COMPONENTS roscpp sensor_msgs std_msgs)
set(SPINNAKER_ROOT "" CACHE PATH "Private vendor SDK prefix")
add_executable(livo_flir_camera main.cpp)
target_include_directories(livo_flir_camera PRIVATE ${catkin_INCLUDE_DIRS} "${SPINNAKER_ROOT}/include")
target_link_libraries(livo_flir_camera ${catkin_LIBRARIES} "${SPINNAKER_ROOT}/lib/libSpinnaker.so")
set_target_properties(livo_flir_camera PROPERTIES BUILD_RPATH "${SPINNAKER_ROOT}/lib" LINK_FLAGS "-Wl,-rpath-link,${SPINNAKER_ROOT}/lib")
